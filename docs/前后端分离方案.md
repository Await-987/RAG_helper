### 3.3 服务层设计

```python
# services/chat_service.py
from core.agents import chat_agent_factory

class ChatService:
    def __init__(self):
        self._agent_cache = {}  # 按 user_id 缓存 agent

    def get_or_create_agent(self, user_id: str):
        """获取或创建用户的 ChatAgent"""
        if user_id not in self._agent_cache:
            self._agent_cache[user_id] = chat_agent_factory()
        return self._agent_cache[user_id]

    async def stream_response(self, user_id: str, message: str):
        """流式响应"""
        agent = self.get_or_create_agent(user_id)

        reasoning = ""
        response = ""

        for chunk in agent.stream_step(message):
            if chunk.get("reasoning"):
                reasoning += chunk["reasoning"]
                yield {"type": "reasoning", "content": chunk["reasoning"]}
            elif chunk.get("content"):
                response += chunk["content"]
                yield {"type": "content", "content": chunk["content"]}

        yield {"type": "done", "reasoning": reasoning, "response": response}

    def clear_conversation(self, user_id: str):
        """清空对话"""
        if user_id in self._agent_cache:
            self._agent_cache[user_id].memory.clear()
```

### 3.4 JWT 认证中间件

```python
# core/security.py
from datetime import datetime, timedelta
from jose import JWTError, jwt
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer

SECRET_KEY = "your-secret-key"
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_HOURS = 24

oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/v1/auth/login")

def create_access_token(data: dict) -> str:
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(hours=ACCESS_TOKEN_EXPIRE_HOURS)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

async def get_current_user(token: str = Depends(oauth2_scheme)) -> UserInfo:
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        username: str = payload.get("sub")
        role: str = payload.get("role")
        if username is None:
            raise credentials_exception
        return UserInfo(username=username, role=role)
    except JWTError:
        raise credentials_exception

async def require_admin(current_user: UserInfo = Depends(get_current_user)) -> UserInfo:
    if current_user.role != "admin":
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="Admin access required"
        )
    return current_user
```

```python
# services/file_service.py
import asyncio
from typing import Dict
import uuid

class ImportTaskManager:
    def __init__(self):
        self.tasks: Dict[str, ImportProgress] = {}

    def create_task(self, file_paths: List[str]) -> str:
        task_id = str(uuid.uuid4())
        self.tasks[task_id] = ImportProgress(
            current=0,
            total=len(file_paths),
            current_file="",
            status="processing"
        )
        # 启动后台任务
        asyncio.create_task(self._run_import(task_id, file_paths))
        return task_id

    async def _run_import(self, task_id: str, file_paths: List[str]):
        from core.tools import load_and_store_file

        for i, file_path in enumerate(file_paths):
            self.tasks[task_id].current = i + 1
            self.tasks[task_id].current_file = os.path.basename(file_path)

            try:
                load_and_store_file(file_path)
            except Exception as e:
                logger.error(f"Import failed: {file_path}, {e}")

        self.tasks[task_id].status = "success"

    def get_progress(self, task_id: str) -> Optional[ImportProgress]:
        return self.tasks.get(task_id)
```

```tsx
    addMessage({ role: 'user', content: message });
    setIsLoading(true);

    // SSE 流式请求
    const eventSource = new EventSource(`/api/v1/chat/stream?message=${encodeURIComponent(message)}`);

    let assistantMessage = { role: 'assistant', content: '', reasoning: '' };

    eventSource.onmessage = (event) => {
      const data = JSON.parse(event.data);

      if (data.type === 'reasoning') {
        assistantMessage.reasoning += data.content;
      } else if (data.type === 'content') {
        assistantMessage.content += data.content;
      } else if (data.type === 'done') {
        addMessage(assistantMessage);
        eventSource.close();
        setIsLoading(false);
      }

      // 实时更新 UI
      updateStreamingMessage(assistantMessage);
    };

    eventSource.onerror = () => {
      eventSource.close();
      setIsLoading(false);
    };
  };

    <div className="flex flex-col h-full">
      <MessageList messages={messages} isLoading={isLoading} />
      <ChatInput onSend={handleSend} disabled={isLoading} />
```

```tsx
import { fileApi } from '@/api/files';
import { FileList } from '@/components/File/FileList';
import { FileUploader } from '@/components/File/FileUploader';

export function FileManager() {
  const [searchQuery, setSearchQuery] = useState('');
  const [filterType, setFilterType] = useState('all');

  // 获取文件列表
  const { data, isLoading, refetch } = useQuery({
    queryKey: ['files', searchQuery, filterType],
    queryFn: () => fileApi.listFiles({ search: searchQuery, filterType }),
  });

  // 导入文件
  const importMutation = useMutation({
    mutationFn: fileApi.importFiles,
    onSuccess: () => refetch(),
  });

  // 删除文件
  const deleteMutation = useMutation({
    mutationFn: fileApi.deleteFile,
    onSuccess: () => refetch(),
  });

  return (
    <div className="p-6">
      {/* 统计卡片 */}
      <div className="grid grid-cols-4 gap-4 mb-6">
        <StatCard label="本地文件" value={data?.stats.local} />
        <StatCard label="已建库" value={data?.stats.imported} />
        <StatCard label="未建库" value={data?.stats.notImported} />
        <StatCard label="残留数据" value={data?.stats.ghost} />
      </div>

      {/* 搜索和筛选 */}
      <div className="flex gap-4 mb-6">
        <Input
          placeholder="搜索文件名..."
          value={searchQuery}
          onChange={(e) => setSearchQuery(e.target.value)}
        />
        <Select value={filterType} onChange={setFilterType}>
          <option value="all">全部</option>
          <option value="imported">已建库</option>
          <option value="not_imported">未建库</option>
          <option value="ghost">残留数据</option>
        </Select>
      </div>

      {/* 上传区域（管理员） */}
      {isAdmin && <FileUploader onUpload={handleUpload} />}

      {/* 文件列表 */}

      />
    </div>
  );
```

```typescript
// stores/chatStore.ts
import { create } from 'zustand';

interface ChatState {
  messages: ChatMessage[];
  streamingMessage: ChatMessage | null;
  addMessage: (message: ChatMessage) => void;
  updateStreamingMessage: (message: Partial<ChatMessage>) => void;
  clearMessages: () => void;
}

export const useChatStore = create<ChatState>((set) => ({
  messages: [],
  streamingMessage: null,

  addMessage: (message) =>
    set((state) => ({
      messages: [...state.messages, message],
      streamingMessage: null,
    })),

  updateStreamingMessage: (partial) =>
    set((state) => ({
      streamingMessage: {
        ...state.streamingMessage,
        ...partial,
      } as ChatMessage,
    })),

  clearMessages: () => set({ messages: [] }),
}));
```

```typescript
// api/client.ts
import axios from 'axios';

const api = axios.create({
  baseURL: '/api/v1',
  timeout: 30000,
});

// 请求拦截器：添加 token
api.interceptors.request.use((config) => {
  const token = localStorage.getItem('access_token');
  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// 响应拦截器：处理错误
api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem('access_token');
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default api;
```

```python
def backend_embedding_model():

    return SentenceTransformerEncoder(...)

    query: str,
```

### 7.2 会话管理

```python
# Streamlit: 使用 st.session_state
if "chat_agent" not in st.session_state:
    st.session_state.chat_agent = chat_agent_factory()

# FastAPI: 使用内存缓存或 Redis
from fastapi_cache import FastAPICache

class SessionManager:
    def __init__(self):
        self._sessions: Dict[str, ChatAgent] = {}

    def get_or_create(self, session_id: str) -> ChatAgent:
        if session_id not in self._sessions:
            self._sessions[session_id] = chat_agent_factory()
        return self._sessions[session_id]
```

### 7.3 流式响应

```python
# Streamlit: 直接使用 st.write_stream
for chunk in agent.stream_step(message):
    st.write(chunk)

# FastAPI: SSE 或 WebSocket
async def stream_generator(message: str):
    agent = get_chat_agent()
    for chunk in agent.stream_step(message):
        yield f"data: {json.dumps(chunk)}\n\n"

@router.post("/stream")
async def stream_chat(request: ChatRequest):
    return StreamingResponse(
        stream_generator(request.message),
        media_type="text/event-stream"
    )
```

## 八、总结

| 对比项 | Streamlit 版本 | FastAPI + React 版本 |
|-------|---------------|---------------------|
| 架构 | 单体应用 | 前后端分离 |
| 扩展性 | 有限 | 易于水平扩展 |
| 用户体验 | 页面刷新 | SPA 流畅体验 |
| 开发效率 | 快速原型 | 需要更多代码 |
| 部署 | 单进程 | 可容器化、负载均衡 |
| 维护性 | 紧耦合 | 关注点分离 |

这个方案能够保持现有核心代码（agents、tools）的最大复用，同时提供更好的用户体验和可扩展性。
